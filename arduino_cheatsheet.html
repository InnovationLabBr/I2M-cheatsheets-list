Blink LED — full example
const int LED = 13;

void setup() {
  pinMode(LED, OUTPUT);
}

void loop() {
  digitalWrite(LED, HIGH);
  delay(500);
  digitalWrite(LED, LOW);
  delay(500);
}
blinks every 0.5s
Common Sensors
// Potentiometer (wiper → A0)
int pot = analogRead(A0); // 0–1023
VCC → left pin, GND → right pin, wiper → A0
// Phototransistor (A0 + 10kΩ to GND)
int light = analogRead(A0);
always use 10kΩ resistor
// Piezo buzzer (pin 8)
tone(8, 440); // 440 Hz
delay(500);
noTone(8);
tone(pin, frequency) — noTone() to stop
Functions — write your own
void blinkLED(int pin, int times) {
  for (int i = 0; i < times; i++) {
    digitalWrite(pin, HIGH); delay(200);
    digitalWrite(pin, LOW); delay(200);
  }
}
// call it:
blinkLED(13, 3);
void = returns nothing
int myFunc() = returns a number
define above or below loop() — both work
